    static func build_ground_image(biome: Dictionary, seed_value: int) -> Image:
    	var rect := ground_rect()
    	var img := Image.create(int(rect.size.x), int(rect.size.y), false, Image.FORMAT_RGBA8)

    	noise.seed = seed_value

```gdscript
class_name Formation
## Bölük konumları. Her ordu kendi yerleşim bölgesinde "bölge koordinatı" kullanır:
## x = cephe boyunca (0..FIELD_WIDTH-1), y = derinlik (0 = ön, rakibe en yakın).
## Rakip ordu aynı koordinatı ayna gibi kullanır (savaş alanının öbür ucunda).
## Otomatik dizilim 3x3 ızgarayı kullanır (cell_to_deploy); oyuncu bölüğü serbestçe taşıyabilir.

const ROW_SPACING := 1.0  ## Izgara satırları arası (karo).
const COL_SPACING := 1.6  ## Izgara sütunları arası (karo).
const DEPLOY_DEPTH := 2.9  ## Bölge derinliği: ızgara sıraları 0, 1, 2 bunun içinde.
const MIN_GAP := 1.0  ## İki bölük merkezi arası en az mesafe (formasyon ~1 karo yayılır).
const FRONT_Y := float(Battlefield.FIELD_DEPTH - Battlefield.DEPLOY_ROWS)


static func cell_to_deploy(cell: Vector2i, grid_size: Vector2i) -> Vector2:
	var center_x := (Battlefield.FIELD_WIDTH - 1) / 2.0
	return Vector2(center_x + (cell.x - (grid_size.x - 1) / 2.0) * COL_SPACING, cell.y * ROW_SPACING)


static func deploy_to_field(pos: Vector2, team: SquadState.Team) -> Vector2:
	var y := FRONT_Y + pos.y
	if team == SquadState.Team.ENEMY:
		y = (Battlefield.FIELD_DEPTH - 1) - y
	return Vector2(pos.x, y)


## Oyuncu tarafı: savaş alanı koordinatı -> bölge koordinatı (sürükleme için).
static func field_to_deploy(p: Vector2) -> Vector2:
	return Vector2(p.x, p.y - FRONT_Y)


static func clamp_deploy(pos: Vector2) -> Vector2:
	return Vector2(clampf(pos.x, 0.0, Battlefield.FIELD_WIDTH - 1), clampf(pos.y, 0.0, DEPLOY_DEPTH))


## pos boşsa kendisi; değilse en yakın boş nokta (halkalarla aranır, deterministik).
static func nearest_free(pos: Vector2, others: Array[Vector2]) -> Vector2:
	pos = clamp_deploy(pos)
	if _is_free(pos, others):
		return pos
	for ring in range(1, 40):
		var radius := ring * 0.25
		var steps := 8 * ring
		for i in steps:
			var p := clamp_deploy(pos + Vector2.from_angle(TAU * i / steps) * radius)
			if _is_free(p, others):
				return p
	return pos


static func _is_free(pos: Vector2, others: Array[Vector2]) -> bool:
	for o in others:
		if pos.distance_to(o) < MIN_GAP - 0.001:
			return false

func _unhandled_input(event: InputEvent) -> void:
	if not visible:
		return
	var field: Battlefield = _meydan.battlefield
	if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT:
		if event.pressed:
			var index := _squad_under(field.mouse_field())
			if index >= 0:
				begin_drag(index)
				get_viewport().set_input_as_handled()
		elif dragging >= 0:
			drop_at(field.mouse_field())
			get_viewport().set_input_as_handled()
	elif event is InputEventMouseMotion:
		var p := field.mouse_field()
		if dragging >= 0:
			_move_view(dragging, Formation.clamp_deploy(Formation.field_to_deploy(p)))
		else:
			var index := _squad_under(p)
			if index >= 0:
				_info.text = SquadInfo.squad_details(RunState.army[index])
				_info.add_theme_color_override("font_color", UiKit.COLOR_TEXT)
	elif event.is_action_pressed("ui_cancel") and dragging >= 0:
		cancel_drag()
		get_viewport().set_input_as_handled()


func begin_drag(index: int) -> void:
	dragging = index
	_drag_from = RunState.army[index].pos


func drop_at(field_pos: Vector2) -> void:
	if dragging < 0:
		return
	var squad: ArmySquad = RunState.army[dragging]
	var pos := RunState.move_squad_to(squad, Formation.field_to_deploy(field_pos))
	_move_view(dragging, pos)
	dragging = -1
	_band_label.visible = false
	Audio.play("card")
	Profile.store_run(RunState.to_dict("prep"))


func cancel_drag() -> void:
	if dragging < 0:
		return
	RunState.army[dragging].pos = _drag_from
	_move_view(dragging, _drag_from)
	dragging = -1
	_band_label.visible = false


## Fare altındaki oyuncu bölüğü (ordu sırası), yoksa -1. Bölük ~1 karo yarıçaplı daire sayılır.
func _squad_under(field_pos: Vector2) -> int:
	var best := -1
	var best_d := 0.6
	for i in RunState.army.size():
		var d := Formation.deploy_to_field(RunState.army[i].pos, SquadState.Team.PLAYER).distance_to(field_pos)
		if d < best_d:
			best = i
			best_d = d
	return best


func _move_view(index: int, deploy_pos: Vector2) -> void:
	var view: SquadView = _meydan.player_views()[index]
	var p := Formation.deploy_to_field(deploy_pos, SquadState.Team.PLAYER)
	view.squad.position = p
	view.squad.previous_position = p
	view.update_visual(1.0, 0.0)
	_band_label.text = UiKit.band_name(ArmySquad.band_of(deploy_pos.y))
	_band_label.visible = dragging >= 0
	_band_label.position = get_viewport().get_mouse_position() + Vector2(10, -14)

func apply_card(offer: CardOffer, replaced: ArmySquad = null) -> void:
	var army_before := RunState.army.duplicate()
	var passives_before := RunState.passives.size()
	RunState.take_card(offer, replaced)
	_show_armies(false)
	var views := player_views()
	for i in RunState.army.size():
		var squad: ArmySquad = RunState.army[i]
		var view: SquadView = views[i]
		if not army_before.has(squad):
			_march_in(view)
		elif squad == RunState.last_changed:
			view.flash()
		elif RunState.passives.size() > passives_before:
			for passive in RunState.passives.slice(passives_before):
				if passive.applies_to(squad):
					view.flash()
	await get_tree().create_timer(0.9).timeout
	_next_battle()


func _march_in(view: SquadView) -> void:
	var end := view.position
	view.position = end + Battlefield.field_offset_to_world(Vector2(0, 4))  # arka kenarın dışından
	view.create_tween().tween_property(view, "position", end, 0.7).set_trans(Tween.TRANS_SINE)


func _next_battle() -> void:
	RunState.advance()
	battlefield.set_biome(RunState.current_opponent().id)
	enter(Phase.PREP)
```
